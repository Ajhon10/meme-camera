cmake_minimum_required(VERSION 3.24)
project(MemeCamNative LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Use Windows 11, MSVC, and Windows SDK 10.0.26100 or newer.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /wd4458 /EHsc /permissive-)
add_library(MemeCamMediaSource SHARED src/mediaSource.cpp src/mediaSource.def)
target_link_libraries(MemeCamMediaSource PRIVATE mfplat mfuuid mf evr mfsensorgroup ole32 propsys advapi32)
add_executable(MemeCamCameraHost src/cameraHost.cpp)
target_link_libraries(MemeCamCameraHost PRIVATE mfplat mfuuid mfsensorgroup mfreadwrite mf ole32 advapi32 shell32)
set_target_properties(MemeCamMediaSource MemeCamCameraHost PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
